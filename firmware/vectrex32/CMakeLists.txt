cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(multitini_vectrex32 LANGUAGES C CXX)
    enable_testing()
endif()
enable_language(CXX)
set(GSBASIC_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../third_party/gsbasic)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/gsbasic_sources.txt" GSBASIC_FILES)
list(TRANSFORM GSBASIC_FILES PREPEND "${GSBASIC_ROOT}/")
add_library(multitini_vectrex32 STATIC ${GSBASIC_FILES} runtime.cpp lowlevel.cpp)
target_include_directories(multitini_vectrex32 PUBLIC ${CMAKE_CURRENT_LIST_DIR}
    PRIVATE ${GSBASIC_ROOT} ${CMAKE_CURRENT_LIST_DIR}/compat)
target_compile_features(multitini_vectrex32 PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(multitini_vectrex32 PRIVATE -fwrapv -fno-strict-aliasing)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    add_executable(vectrex32_host host.cpp)
    target_link_libraries(vectrex32_host PRIVATE multitini_vectrex32)
    add_executable(vectrex32_tests tests.cpp)
    target_link_libraries(vectrex32_tests PRIVATE multitini_vectrex32)
    add_test(NAME vectrex32_runtime COMMAND vectrex32_tests)
    add_executable(vectrex32_lowlevel_tests lowlevel_tests.cpp)
    target_link_libraries(vectrex32_lowlevel_tests PRIVATE multitini_vectrex32)
    add_test(NAME vectrex32_lowlevel COMMAND vectrex32_lowlevel_tests)
endif()
